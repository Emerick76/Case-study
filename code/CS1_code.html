This notebook produces every table, figure and number in my report on the case Robinhood Traders and the Case of GameStop. Run it from top to bottom (Kernel → Restart & Run All). It reads the four CSV files in this folder and writes Tables 1–6 to tables/, Figures 1–6 to figures/ and the numbers quoted in the report to tables/report_numbers.json.

Data:

  • RH_data.csv: Robinhood holdings (number of users holding each ticker, daily), provided with the case.
  • stock_data.csv: CRSP daily stock file (Center for Research in Security Prices) through WRDS: daily return with dividends (DlyRet) of the Robinhood tickers, and the CRSP value-weighted market return (vwretd).
  • sp500_yahoo.csv: S&P 500 index (^GSPC) and SPY daily returns from Yahoo Finance (Microsoft Qlib US dataset).
  • gme_yahoo.csv: GameStop (GME) daily prices from Yahoo Finance, 2002–2022 (us_stock_eod repository). build_market_files.py shows how the two Yahoo files were built.

Packages: numpy, pandas, scipy, statsmodels, matplotlib (pip install -r requirements.txt).

# --------------------------------------------------------------------------------------
# (b)-(c) The representative investor holds ticker i in proportion to the number of users holding it:
#     w_{i,t} = users_{i,t} / sum_j users_{j,t}
# (d) Portfolio return on trading day t: R_t = sum_i w_{i,t-1} r_{i,t}.
#     Weights are taken from the last RH date BEFORE day t (the holdings at the start of the day), so the portfolio
#     does not use trades made after that day's price moves. The RH file has calendar days, so on a Monday the
#     weights are Sunday's holdings; for the days missing from the file the last available day is used.
#     On each day, tickers without a CRSP return drop out and the remaining weights are rescaled to sum to one.
# Robustness: the same portfolio with same-day weights w_{i,t} (this mixes in look-ahead).
# --------------------------------------------------------------------------------------
def holdings_for_days(H, days, lag=True):
    """Holdings row used on each trading day: last RH date strictly before the day (lag=True) or on or before
    it (lag=False). Returns the (day x ticker) table and the age of the holdings in calendar days."""
    pos = H.index.searchsorted(days, side="left" if lag else "right") - 1
    if (pos < 0).any():
        raise ValueError("no RH holdings before the first trading day")
    Hd = H.iloc[pos].copy()
    Hd.index = days
    return Hd, pd.Series((days - H.index[pos]).days, index=days)


def rh_portfolio(H, R, lag=True):
    """Weights, daily return, coverage (share of users with a CRSP return) and holdings age of the RH portfolio."""
    Hd, age = holdings_for_days(H, R.index, lag)
    Rr = R.reindex(columns=H.columns)
    users = Hd.where(Rr.notna())                       # drop tickers without a return that day
    W = users.div(users.sum(axis=1), axis=0)
    ret = (W * Rr).sum(axis=1, min_count=1)
    return ret, W, users.sum(axis=1) / Hd.sum(axis=1), age

def max_drawdown(r):
    """Largest peak-to-trough fall of the value of $1 invested."""
    w = (1 + r).cumprod()
    return (w / w.cummax() - 1).min()


def summary_stats(rets, bench=BENCH, k=4):
    """Table 2: descriptive statistics (moments as in Lecture 1: scipy skew and Fisher kurtosis), performance
    measures, and an OLS of each series on the benchmark (market model, HC1 standard errors)."""
    z = (rets - rets.mean()) / rets.std()
    out = {}
    for s in rets:
        r = rets[s]
        row = {"mean (daily)": r.mean(), "std. dev. (daily)": r.std(),
               "annualised mean": r.mean() * ANN, "annualised volatility": r.std() * np.sqrt(ANN),
               "minimum": r.min(), "maximum": r.max(),
               "skewness": stats.skew(r), "excess kurtosis": stats.kurtosis(r),
               f"days with |z| > {k}": int((z[s].abs() > k).sum()),
               "cumulative return": (1 + r).prod() - 1, "maximum drawdown": max_drawdown(r),
               "Sharpe ratio (annualised, rf = 0)": r.mean() / r.std() * np.sqrt(ANN)}
        if s != bench:
            ols = sm.OLS(r, sm.add_constant(rets[bench])).fit(cov_type="HC1")
            row.update({"correlation with S&P 500": r.corr(rets[bench]), "beta on S&P 500": ols.params.iloc[1],
                        "alpha (annualised)": ols.params.iloc[0] * ANN, "alpha t-stat (HC1)": ols.tvalues.iloc[0],
                        "R-squared": ols.rsquared})
        out[s] = row
    return pd.DataFrame(out)

# --------------------------------------------------------------------------------------
# Historical-simulation VaR and ES (Lecture 2, slides 10-11 and A.8), the same functions as in Homework 1.
# Normal VaR and ES for comparison (Lecture 2): VaR = -(mu + sigma z_a), ES = -mu + sigma phi(z_a) / a.
# Sampling uncertainty: i.i.d. bootstrap, 10,000 resamples, 95% percentile interval (Lecture 1, A.22).
# --------------------------------------------------------------------------------------
def empirical_var(r, alpha, axis=-1):
    """Historical-simulation VaR at tail probability alpha, as a positive loss: VaR = -q_alpha(r)."""
    return -np.quantile(r, alpha, axis=axis, method="interpolated_inverted_cdf")


def var_es_empirical(r, alpha):
    """Empirical VaR and ES (minus the mean of the floor(alpha N) lowest returns), positive losses."""
    r = np.asarray(r, dtype=float)
    k = int(np.floor(alpha * r.size))
    if k < 1:
        raise ValueError("alpha * N < 1: too few observations for an empirical ES")
    return empirical_var(r, alpha), -np.sort(r)[:k].mean()


def var_es_normal(r, alpha):
    """Normal VaR and ES with the sample mean and standard deviation, positive losses."""
    mu, sd, za = np.mean(r), np.std(r, ddof=1), stats.norm.ppf(alpha)
    return -(mu + sd * za), -mu + sd * stats.norm.pdf(za) / alpha


def bootstrap_var_es(rng, r, alpha, B=10_000, chunk=1_000):
    """B i.i.d. bootstrap samples of r; returns the B empirical VaRs and ESs."""
    r = np.asarray(r, dtype=float)
    k = int(np.floor(alpha * r.size))
    v, e = [], []
    for _ in range(B // chunk):
        S = rng.choice(r, size=(chunk, r.size), replace=True)
        v.append(empirical_var(S, alpha, axis=1))
        e.append(-np.sort(S, axis=1)[:, :k].mean(axis=1))
    return np.concatenate(v), np.concatenate(e)

# --------------------------------------------------------------------------------------
# Out-of-sample check: rolling historical VaR on the previous 250 trading days, tested on the next day, which
# covers 2020 (COVID-19 crash). Coverage test as in Lecture 2 / Christoffersen ch. 8: hits ~ Binomial(T, alpha)
# under H0, exact two-sided p-value.
# --------------------------------------------------------------------------------------
VAR_WINDOW = 250


def rolling_hs_var(r, alpha, window=VAR_WINDOW):
    """VaR_t from the `window` returns before day t (historical simulation), for every day after the first window."""
    v = r.to_numpy()
    return pd.Series([empirical_var(v[k - window:k], alpha) for k in range(window, v.size)], index=r.index[window:])


def coverage_test(r, var, p):
    """Hits I_t = 1(r_t < -VaR_t); under H0 the number of hits is Binomial(T, p). Exact two-sided p-value."""
    hits = (r < -var).to_numpy()
    T, s1 = hits.size, int(hits.sum())
    return {"days tested": T, "hits": s1, "expected hits": p * T, "hit rate": s1 / T,
            "binomial p-value": stats.binomtest(s1, T, p).pvalue}

# --------------------------------------------------------------------------------------
# Lag order (Lecture 4): AIC and BIC of AR(p), p = 0..10, by OLS on a common sample, using only data BEFORE the
# forecast period (2002-2018), so that the choice of p does not use the data being forecast.
# --------------------------------------------------------------------------------------
FC_START = "2019-01-02"            # first forecast date
ROLL_R = 250                       # rolling window, in trading days (about one year)
PMAX = 10


def lag_matrix(v, p, start):
    """Regressors [1, v_{t-1}, ..., v_{t-p}] and target v_t for t = start, ..., len(v)-1."""
    n = v.size
    X = np.column_stack([np.ones(n - start)] + [v[start - i:n - i] for i in range(1, p + 1)])
    return X, v[start:]


def ols_fit(X, t):
    """OLS coefficients, residual variance (dof-corrected) and coefficient covariance."""
    beta, *_ = np.linalg.lstsq(X, t, rcond=None)
    e = t - X @ beta
    s2 = e @ e / (t.size - X.shape[1])
    return beta, s2, s2 * np.linalg.inv(X.T @ X), e

# --------------------------------------------------------------------------------------
# Rolling one-day-ahead forecasts (Lecture 5). For every forecast date t from 2 Jan 2019 to the end of the data:
#   1. fit AR(p) by OLS on the ROLL_R returns before t (y_{t-R}, ..., y_{t-1}), with a constant;
#   2. forecast y_t = c + phi_1 y_{t-1} + ... + phi_p y_{t-p}, with a 95% interval forecast +/- 1.96 sigma_hat
#      (sigma_hat: residual s.d. of that window; parameter uncertainty ignored);
#   3. benchmarks: the window's mean return (historical mean) and zero (random walk for the log price).
# --------------------------------------------------------------------------------------
def rolling_ar_forecasts(y, p, R, start):
    v, idx = y.to_numpy(), y.index
    k0 = idx.searchsorted(pd.Timestamp(start))
    if k0 < R:
        raise ValueError("not enough data before the first forecast")
    rows = []
    for k in range(k0, v.size):
        win = v[k - R:k]
        X, t = lag_matrix(win, p, p)
        beta, s2, cov, _ = ols_fit(X, t)
        x_next = np.r_[1.0, v[k - 1:k - p - 1:-1] if k - p - 1 >= 0 else v[k - 1::-1]]
        rows.append({"date": idx[k], "actual": v[k], "forecast AR": x_next @ beta, "sigma": np.sqrt(s2),
                     "forecast mean": win.mean(), "phi_1": beta[1], "se phi_1": np.sqrt(cov[1, 1]),
                     "sum of phi": beta[1:].sum()})
    f = pd.DataFrame(rows).set_index("date")
    f["lower 95%"] = f["forecast AR"] - 1.96 * f["sigma"]
    f["upper 95%"] = f["forecast AR"] + 1.96 * f["sigma"]
    return f

# --------------------------------------------------------------------------------------
# Accuracy (Lecture 5): RMSFE and MAE; Theil's U = RMSFE(AR) / RMSFE(zero forecast); out-of-sample R2 =
# 1 - MSE(AR) / MSE(historical mean); Mincer-Zarnowitz regression
# y_t = a + b f_t + u_t with a Wald test of (a, b) = (0, 1) (HC1); direction (sign) hit rate with a binomial test
# against 50%; coverage of the 95% interval forecast; modified Diebold-Mariano test (MDM, squared loss, h = 1 so
# no Newey-West lags). Periods: before the event, the event window (from the new directors on 11 Jan 2021 to
# 5 Feb 2021, Thorbecke 2021), after it.
# --------------------------------------------------------------------------------------
PERIODS_3 = {"Full period": (FC_START, None), "Before: 2 Jan 2019 – 8 Jan 2021": (FC_START, "2021-01-08"),
             "Event: 11 Jan – 5 Feb 2021": ("2021-01-11", "2021-02-05"),
             "After: 8 Feb 2021 – 25 Mar 2022": ("2021-02-08", None)}


def evaluate(f):
    """Forecast accuracy measures for one period of the rolling forecasts."""
    e, e_mean, e_zero = f["actual"] - f["forecast AR"], f["actual"] - f["forecast mean"], f["actual"]
    rmse = lambda x: np.sqrt(np.mean(x ** 2))
    mz = sm.OLS(f["actual"].to_numpy(), sm.add_constant(f["forecast AR"].to_numpy())).fit(cov_type="HC1")
    wald = mz.wald_test((np.eye(2), np.array([0.0, 1.0])), scalar=True)
    nz = (np.sign(f["forecast AR"]) != 0) & (f["actual"] != 0)
    hits = int((np.sign(f["forecast AR"][nz]) == np.sign(f["actual"][nz])).sum())
    inside = (e.abs() <= 1.96 * f["sigma"])
    return {"days": len(f), "actual: mean": f["actual"].mean(), "actual: s.d.": f["actual"].std(),
            "forecast: mean": f["forecast AR"].mean(), "forecast: s.d.": f["forecast AR"].std(),
            "forecast: min": f["forecast AR"].min(), "forecast: max": f["forecast AR"].max(),
            "RMSFE AR": rmse(e), "RMSFE historical mean": rmse(e_mean), "RMSFE zero": rmse(e_zero),
            "Theil's U (AR / zero)": rmse(e) / rmse(e_zero),
            "out-of-sample R2 (vs mean)": 1 - np.mean(e ** 2) / np.mean(e_mean ** 2), "MAE AR": e.abs().mean(),
            "MZ intercept a": mz.params[0], "MZ slope b": mz.params[1], "MZ p-value (a, b) = (0, 1)": float(wald.pvalue),
            "sign hit rate": hits / int(nz.sum()), "sign p-value (vs 50%)": stats.binomtest(hits, int(nz.sum()), 0.5).pvalue,
            "95% interval coverage": inside.mean(), "days outside the 95% interval": int((~inside).sum())}


def mdm_test(e1, e2, h=1):
    """Modified Diebold-Mariano test (Harvey, Leybourne and Newbold 1997) of equal squared loss, two-sided.
    d_t = e1_t^2 - e2_t^2 (negative mean: model 1 better). Newey-West variance with h-1 lags."""
    d = (np.asarray(e1) ** 2 - np.asarray(e2) ** 2)
    n, dbar = d.size, d.mean()
    gam = [np.mean((d[j:] - dbar) * (d[:n - j] - dbar)) for j in range(h)]
    var = (gam[0] + 2 * sum(gam[1:])) / n
    mdm = dbar / np.sqrt(var) * np.sqrt((n + 1 - 2 * h + h * (h - 1) / n) / n)
    return dbar, mdm, 2 * stats.t.sf(abs(mdm), n - 1)
